### 2.4 语音模态编码器

```python
# 架构：Bi-LSTM + Mean Pooling → 512维特征向量
class VoiceEncoder(nn.Module):
    def __init__(self):
        super().__init__()
        # 声学特征路径
        self.acoustic_proj = nn.Linear(88, 256)
        self.lstm = nn.LSTM(256, 128, num_layers=2, bidirectional=True, batch_first=True)
        # 文本特征路径（BERT输出768维）
        self.text_proj = nn.Linear(768, 256)
        # 融合
        self.fusion = nn.Linear(256 + 256, 512)
        self.norm = nn.LayerNorm(512)

    def forward(self, acoustic_seq, bert_embedding):
        # acoustic_seq: (batch, time_steps, 88)
        acoustic = self.acoustic_proj(acoustic_seq)
        lstm_out, _ = self.lstm(acoustic)  # (batch, time_steps, 256)
        acoustic_feat = lstm_out.mean(dim=1)  # Mean Pooling → (batch, 256)
        # 文本特征
        text_feat = self.text_proj(bert_embedding)  # (batch, 256)
        # 拼接融合
        combined = torch.cat([acoustic_feat, text_feat], dim=-1)  # (batch, 512)
        return self.norm(self.fusion(combined))  # (batch, 512)
```

## 五、核心创新：跨模态注意力融合层

### 5.1 设计动机

传统多模态融合方案的局限性：
- **早期融合（特征拼接）**：简单，但忽略模态间相互关系，高维度导致过拟合
- **晚期融合（结果投票）**：损失跨模态交互信息，各模态独立判断再汇总
- **本系统方案**：Cross-Modal Attention，允许每个模态"询问"其他模态的信息，学习模态间的互补关系

### 5.2 算法实现

```python
class CrossModalAttentionFusion(nn.Module):
    """
    跨模态注意力融合
    输入：4个模态的特征向量 [voice(512), visual(512), behavior(256), scale(128)]
    输出：融合特征向量 (1024维)
    """
    def __init__(self, d_model=256, n_heads=4):
        super().__init__()
        # 将各模态投影到统一维度d_model=256
        self.voice_proj    = nn.Linear(512, d_model)
        self.visual_proj   = nn.Linear(512, d_model)
        self.behavior_proj = nn.Linear(256, d_model)
        self.scale_proj    = nn.Linear(128, d_model)

        # 多头交叉注意力（每个模态作为Query，其余模态作为K/V）
        self.cross_attn = nn.MultiheadAttention(d_model, n_heads, batch_first=True)

        # 残差连接与归一化
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)

        # 前馈网络
        self.ffn = nn.Sequential(
            nn.Linear(d_model, d_model * 4),
            nn.GELU(),
            nn.Dropout(0.1),
            nn.Linear(d_model * 4, d_model)
        )

        # 最终分类头
        self.classifier = nn.Sequential(
            nn.Linear(d_model * 4, 512),  # 4个模态各256维拼接
            nn.GELU(),
            nn.Dropout(0.3),
            nn.Linear(512, 128),
            nn.GELU(),
            nn.Linear(128, 3)  # 3类：NC/MCI/mild_AD
        )

    def forward(self, voice_feat, visual_feat, behavior_feat, scale_feat,
                return_attn_weights=False):
        # Step 1: 投影到统一维度
        v = self.voice_proj(voice_feat).unsqueeze(1)      # (B, 1, 256)
        vi = self.visual_proj(visual_feat).unsqueeze(1)   # (B, 1, 256)
        b = self.behavior_proj(behavior_feat).unsqueeze(1) # (B, 1, 256)
        s = self.scale_proj(scale_feat).unsqueeze(1)      # (B, 1, 256)

        # Step 2: 所有模态作为一个序列
        modalities = torch.cat([v, vi, b, s], dim=1)  # (B, 4, 256)

        # Step 3: 自注意力（让每个模态关注其他模态）
        attn_out, attn_weights = self.cross_attn(modalities, modalities, modalities)

        # Step 4: 残差 + 归一化
        modalities = self.norm1(modalities + attn_out)
        ffn_out = self.ffn(modalities)
        modalities = self.norm2(modalities + ffn_out)  # (B, 4, 256)

        # Step 5: 展平拼接
        fused = modalities.view(modalities.size(0), -1)  # (B, 1024)

        # Step 6: 分类
        logits = self.classifier(fused)  # (B, 3)

        if return_attn_weights:
            return logits, attn_weights, fused  # 返回注意力权重用于SHAP解释
        return logits
```

### 5.3 注意力权重的可解释性

注意力权重矩阵 `attn_weights`（4×4）展示了每个模态"关注"其他模态的程度：

```
语音模态关注度 = attn_weights[0, :]
→ [0.38, 0.22, 0.25, 0.15] 表示语音模态主要依靠自身(0.38)和视觉信息(0.22)
```

结合SHAP进一步量化特征贡献：
```python

# 2. Focal Loss（深度模型训练时聚焦难分类样本）
class FocalLoss(nn.Module):
    def __init__(self, gamma=2.0, alpha=[0.3, 0.3, 0.4]):
        super().__init__()
        self.gamma = gamma
        self.alpha = torch.tensor(alpha)

    def forward(self, logits, targets):
        ce_loss = F.cross_entropy(logits, targets, reduction='none')
        pt = torch.exp(-ce_loss)
        focal_loss = self.alpha[targets] * (1 - pt) ** self.gamma * ce_loss
        return focal_loss.mean()
```

## 七、纵向风险预测模型

### 7.1 模型架构

```python
class LongitudinalRiskPredictor(nn.Module):
    """
    基于用户历次评估数据预测MCI→AD转化风险
    输入：用户过去N次评估的融合特征序列 (N, feature_dim)
    输出：12个月内转化概率 P ∈ [0, 1]
    """
    def __init__(self, feature_dim=1024, hidden_size=256):
        super().__init__()
        self.lstm = nn.LSTM(
            input_size=feature_dim,
            hidden_size=hidden_size,
            num_layers=2,
            bidirectional=True,
            batch_first=True,
            dropout=0.3
        )
        self.attention = nn.Linear(hidden_size * 2, 1)
        self.classifier = nn.Sequential(
            nn.Linear(hidden_size * 2, 64),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(64, 1),
            nn.Sigmoid()
        )

    def forward(self, feature_seq):
        # feature_seq: (B, N, feature_dim)
        lstm_out, _ = self.lstm(feature_seq)  # (B, N, 512)
        # Attention加权
        attn_scores = self.attention(lstm_out).squeeze(-1)  # (B, N)
        attn_weights = F.softmax(attn_scores, dim=1)
        context = (lstm_out * attn_weights.unsqueeze(-1)).sum(dim=1)  # (B, 512)
        risk_prob = self.classifier(context)  # (B, 1)
        return risk_prob
```

**触发条件：** 用户累计评估次数 ≥ 2次

```python
def content_based_recommend(user_domain_scores: dict, intervention_library: list, top_k=5):
    """
    基于用户薄弱认知域推荐干预方案
    user_domain_scores: {"memory": 45, "attention": 70, "executive": 60, ...}
    """
    # 计算各认知域的"需求权重"（分数越低，权重越高）
    demand_weights = {
        domain: max(0, (100 - score) / 100)
        for domain, score in user_domain_scores.items()
    }
    # Softmax归一化
    total = sum(demand_weights.values())
    demand_vector = [demand_weights.get(d, 0) / total
                     for d in ['memory','attention','executive','language','visuospatial','orientation']]

    # 计算每个干预方案与需求的匹配度（余弦相似度）
    scored_interventions = []
    for item in intervention_library:
        coverage_vector = item['domain_coverage']  # 方案覆盖认知域的向量
        similarity = cosine_similarity(demand_vector, coverage_vector)
        scored_interventions.append((item, similarity))

    # 返回Top-K
    return sorted(scored_interventions, key=lambda x: x[1], reverse=True)[:top_k]
```

## 九、模型训练策略

### 9.1 数据集来源

| 数据集 | 类型 | 用途 |
|--------|------|------|
| DementiaBank Pitt Corpus | 语音对话（英文，需翻译适配） | 语音模态预训练 |
| ADNI（阿尔茨海默病神经影像学计划） | 纵向认知评估 | 风险预测模型训练 |
| 自采数据（社区老人） | 多模态评估数据 | 全模型微调 |
| 合成数据（SMOTE增强） | 少数类样本 | 解决类别不平衡 |

### 9.2 训练流程

```
Step 1: 各模态编码器独立预训练
  - VoiceEncoder：在DementiaBank语音数据上预训练
  - VisualEncoder：在FER2013数据上预训练面部表情分类

Step 2: 端到端联合微调（自采多模态数据）
  - 冻结预训练编码器权重 → 训练融合层 → 解冻全部权重 → 端到端微调

Step 3: 知识蒸馏（可选，模型压缩用于移动端部署）
  - Teacher: 完整多模态模型
  - Student: 仅量表+行为特征的轻量模型
```

### 9.3 评估指标

| 指标 | 目标值 | 说明 |
|------|--------|------|
| F1-macro（三分类） | ≥ 0.80 | NC/MCI/mild_AD三类均衡考量 |
| AUC（NC vs 异常） | ≥ 0.90 | 筛查能力 |
| Sensitivity（MCI检出率） | ≥ 0.85 | 宁可误报不漏报 |
| Specificity | ≥ 0.75 | 控制假阳性率 |
| 风险预测AUC | ≥ 0.75 | 纵向转化预测 |
